### **Quarantine / Trimming Logic Hook – Step-by-Step Implementation**

> **Goal:** Automatically **exclude or heavily down-weight** clients whose trust score falls below a threshold τ for *Q* consecutive rounds, while still using the trust-weighted trimmed-mean for the remaining clients.

#### **2. Persistent State Helper (optional, keeps code tidy)**

```python
# trust_module/state.py
from collections import defaultdict
from dataclasses import dataclass

@dataclass
class ClientStatus:
    below_tau_streak: int = 0
    quarantine_left: int = 0

class TrustState:
    """Singleton-style container for per-client status."""
    def __init__(self):
        self._status = defaultdict(ClientStatus)

    def step(self, client_id: str, trust: float, tau: float, patience: int, quarantine: int):
        st = self._status[client_id]
        # update streak
        if trust < tau:
            st.below_tau_streak += 1
        else:
            st.below_tau_streak = 0
        # enter quarantine
        if st.below_tau_streak >= patience:
            st.quarantine_left = quarantine
            st.below_tau_streak = 0
        # tick down existing quarantine
        if st.quarantine_left > 0:
            st.quarantine_left -= 1
        self._status[client_id] = st  # save back

    def is_quarantined(self, client_id: str) -> bool:
        return self._status[client_id].quarantine_left > 0
```

Instantiate one global `trust_state = TrustState()` inside `trust_evaluator.py`.

#### **3. Finish `detect_malicious_clients`**

```python
# trust_module/trust_evaluator.py  (inside TrustEvaluator class)
from trust_module.state import trust_state

def detect_malicious_clients(
    self,
    client_ids: list[str],
    trust_vec: list[float],
) -> tuple[list[str], list[str]]:
    """
    Returns:
      - quarantined: list of client_ids to drop this round
      - survivors:   list to keep for aggregation
    """
    cfg = self.cfg["trust"]            # loaded YAML
    tau      = cfg["tau"]
    patience = cfg["patience"]
    quarantine = cfg["quarantine_rounds"]

    quarantined, survivors = [], []
    for cid, t in zip(client_ids, trust_vec):
        trust_state.step(cid, t, tau, patience, quarantine)
        if trust_state.is_quarantined(cid):
            quarantined.append(cid)
        else:
            survivors.append(cid)
    return quarantined, survivors
```

#### **4. Plug Quarantine into `aggregate_model_updates`**

```python
# still in TrustEvaluator.aggregate_model_updates
trust_vec = self._compute_trust_scores(weights_list, metrics_list, round_idx)

quarantined, survivors = self.detect_malicious_clients(client_ids, trust_vec)

# filter all lists by survivors’ indices
keep_idx = [i for i, cid in enumerate(client_ids) if cid in survivors]
weights_kept   = [weights_list[i] for i in keep_idx]
trust_kept_vec = [trust_vec[i]    for i in keep_idx]

# ---------- trimmed-mean on survivors ----------
trim_k = int(len(weights_kept) * self.cfg["trust"]["trim_ratio"])
agg_weights = self._trimmed_mean(weights_kept, trust_kept_vec, trim_k)
return agg_weights, trust_vec          # return full vec for logging
```

```python
def test_quarantine_cycle():
    cfg = load_yaml("config/trust.yaml")
    te  = TrustEvaluator(cfg)
    low  = [[np.zeros(1)]]  # dummy weights
    high = [[np.ones(1)]]

    client_ids = ["bad", "good"]
    # 3 rounds where 'bad' is below tau
    for r in range(3):
        agg, _ = te.aggregate_model_updates(
            weights_list=low + high,
            client_ids=client_ids,
            metrics_list=[{}, {}],
            round_idx=r,
        )
    # now 'bad' should be quarantined
    assert te.detect_malicious_clients(client_ids, [0.1, 0.9])[0] == ["bad"]
```

### **Done**

You now have:

* **Automatic malicious-client detection** using sustained low trust.
* **Soft ban** for *Q* rounds, after which the client can try to re-enter if its behaviour improves.
* **Trimmed-mean aggregation** applied **after** dropping the quarantined set – protects against residual outliers.
